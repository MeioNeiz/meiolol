--- build.zig
const std = @import("std");

pub fn build(b: *std.Build) void {
    const target = b.standardTargetOptions(.{});
    const optimize = b.standardOptimizeOption(.{});
    const dvui_dep = b.dependency("dvui", .{ .target = target, .optimize = optimize, .backend = .dx11, .@"tree-sitter" = false });
    const exe = b.addExecutable(.{
        .name = "hello",
        .root_module = b.createModule(.{
            .root_source_file = b.path("src/main.zig"),
            .target = target,
            .optimize = optimize,
        }),
    });
    exe.root_module.addImport("dvui", dvui_dep.module("dvui_dx11"));
    exe.subsystem = .Windows;
    b.installArtifact(exe);
}

const std = @import("std");
const dvui = @import("dvui");

pub const dvui_app: dvui.App = .{
    .config = .{ .options = .{ .size = .{ .w = 640.0, .h = 480.0 }, .title = "hello" } },
    .frameFn = frame,
};
pub const main = dvui.App.main;
pub const panic = dvui.App.panic;
pub const std_options: std.Options = .{ .logFn = dvui.App.logFn };

const xs = [_]f64{ 0, 5, 10, 15, 20, 25, 30 };
const ys = [_]f64{ 50, 51, 49.5, 52, 53, 52.5, 54 };

fn frame() !dvui.App.Result {
    var box = dvui.box(@src(), .{}, .{ .expand = .both });
    defer box.deinit();
    dvui.label(@src(), "Win rate by minute", .{}, .{});
    dvui.plotXY(@src(), .{ .xs = &xs, .ys = &ys }, .{ .expand = .both });
    return .ok;
}

const std = @import("std");
const w = @import("win32").everything;

fn wndProc(hwnd: w.HWND, msg: u32, wp: w.WPARAM, lp: w.LPARAM) callconv(.winapi) w.LRESULT {
    if (msg == w.WM_DESTROY) {
        w.PostQuitMessage(0);
        return 0;
    }
    return w.DefWindowProcW(hwnd, msg, wp, lp);
}

pub fn main() void {
    const class_name = std.unicode.utf8ToUtf16LeStringLiteral("hello");
    const wc: w.WNDCLASSEXW = .{
        .cbSize = @sizeOf(w.WNDCLASSEXW),
        .style = .{},
        .lpfnWndProc = wndProc,
        .cbClsExtra = 0,
        .cbWndExtra = 0,
        .hInstance = null,
        .hIcon = null,
        .hCursor = null,
        .hbrBackground = null,
        .lpszMenuName = null,
        .lpszClassName = class_name,
        .hIconSm = null,
    };
    _ = w.RegisterClassExW(&wc);
    const style: w.WINDOW_STYLE = @bitCast(@as(u32, @bitCast(w.WS_OVERLAPPEDWINDOW)) | @as(u32, @bitCast(w.WS_VISIBLE)));
    _ = w.CreateWindowExW(.{}, class_name, class_name, style, w.CW_USEDEFAULT, w.CW_USEDEFAULT, 640, 480, null, null, null, null);
    var msg: w.MSG = undefined;
    while (w.GetMessageW(&msg, null, 0, 0) > 0) {
        _ = w.TranslateMessage(&msg);
        _ = w.DispatchMessageW(&msg);
    }
}
